<script>
// Khởi tạo danh sách giao dịch từ LocalStorage hoặc mảng rỗng
let transactions = JSON.parse(localStorage.getItem('truong_thu_chi_data')) || [
{ id: 1, description: 'Lương tháng', amount: 15000000, type: 'income', category: 'Lương', date: new Date().toLocaleDateString('vi-VN') },
{ id: 2, description: 'Ăn trưa văn phòng', amount: 55000, type: 'expense', category: 'Ăn uống', date: new Date().toLocaleDateString('vi-VN') },
{ id: 3, description: 'Đổ xăng xe máy', amount: 70000, type: 'expense', category: 'Di chuyển', date: new Date().toLocaleDateString('vi-VN') }
];

// Format tiền tệ VND
function formatVND(number) {
return new Intl.NumberFormat('vi-VN', { style: 'currency', currency: 'VND' }).format(number);
}

// Cập nhật giao diện tổng số dư, thu, chi liên tục
function updateDashboard() {
let totalIncome = 0;
let totalExpense = 0;

transactions.forEach(item => {
if (item.type === 'income') {
totalIncome += Number(item.amount);
} else {
totalExpense += Number(item.amount);
}
});

const balance = totalIncome - totalExpense;

document.getElementById('totalIncome').innerText = formatVND(totalIncome);
document.getElementById('totalExpense').innerText = formatVND(totalExpense);

const balanceEl = document.getElementById('totalBalance');
balanceEl.innerText = formatVND(balance);

if (balance < 0) {
balanceEl.className = "text-3xl font-extrabold tracking-tight mt-1 text-rose-400";
} else {
balanceEl.className = "text-3xl font-extrabold tracking-tight mt-1 text-white";
}

document.getElementById('transactionCount').innerText = `${transactions.length} giao dịch`;

// Lưu vào source (LocalStorage)
localStorage.setItem('truong_thu_chi_data', JSON.stringify(transactions));

renderTransactionList();
}

// Hiển thị danh sách giao dịch ra màn hình
function renderTransactionList() {
const listEl = document.getElementById('transactionList');
listEl.innerHTML = '';

if (transactions.length === 0) {
listEl.innerHTML = `
<div class="text-center py-8 text-slate-500 text-xs">
Chưa có giao dịch nào được ghi nhận. Hãy nhập vào ô bên trên!
</div>
`;
return;
}

// Sắp xếp mới nhất lên đầu
const sortedTransactions = [...transactions].reverse();

sortedTransactions.forEach(item => {
const isIncome = item.type === 'income';
const div = document.createElement('div');
div.className = "bg-slate-800/40 border border-slate-700/50 rounded-2xl p-3.5 flex items-center justify-between hover:bg-slate-800 transition";
div.innerHTML = `
<div class="flex items-center space-x-3">
<div class="w-9 h-9 rounded-xl ${isIncome ? 'bg-emerald-500/20 text-emerald-400' : 'bg-rose-500/20 text-rose-400'} flex items-center justify-center text-sm font-bold">
<i class="fa-solid ${isIncome ? 'fa-arrow-down' : 'fa-arrow-up'}"></i>
</div>
<div>
<h4 class="text-xs font-bold text-white">${escapeHtml(item.description)}</h4>
<div class="flex items-center space-x-2 mt-0.5">
<span class="text-[10px] text-slate-400 bg-slate-700/50 px-1.5 py-0.5 rounded">${item.category}</span>
<span class="text-[10px] text-slate-500">${item.date}</span>
</div>
</div>
</div>
<div class="flex items-center space-x-3">
<span class="text-xs font-bold ${isIncome ? 'text-emerald-400' : 'text-slate-100'}">
${isIncome ? '+' : '-'}${formatVND(item.amount)}
</span>
<button onclick="deleteTransaction(${item.id})" class="text-slate-500 hover:text-rose-400 transition text-xs p-1" title="Xóa">
<i class="fa-solid fa-xmark"></i>
</button>
</div>
`;
listEl.appendChild(div);
});
}

// Thêm giao dịch mới
function addTransaction(e) {
e.preventDefault();
const desc = document.getElementById('description').value.trim();
const amount = parseFloat(document.getElementById('amount').value);
const type = document.getElementById('type').value;
const category = document.getElementById('category').value;

if (!desc || isNaN(amount) || amount <= 0) return;

const newTx = {
id: Date.now(),
description: desc,
amount: amount,
type: type,
category: category,
date: new Date().toLocaleDateString('vi-VN')
};

transactions.push(newTx);

// Reset form
document.getElementById('transactionForm').reset();

// Cập nhật ngay lập tức giao diện
updateDashboard();
}

// Xóa một giao dịch
function deleteTransaction(id) {
transactions = transactions.filter(item => item.id !== id);
updateDashboard();
}

// Xóa sạch dữ liệu
function clearData() {
if (confirm("Bạn có chắc chắn muốn xóa sạch toàn bộ lịch sử thu chi không?")) {
transactions = [];
updateDashboard();
}
}

// Xuất file PDF sao kê danh sách thu chi chuẩn chuyên nghiệp
function exportToPDF() {
const { jsPDF } = window.jspdf;
const doc = new jsPDF();

// Tiêu đề báo cáo
doc.setFont("helvetica", "bold");
doc.setFontSize(18);
doc.text("SAO KE THU CHI CA NHAN", 14, 20);

doc.setFontSize(11);
doc.setFont("helvetica", "normal");
doc.text(`Chu tai khoan: DOAN NHUT TRUONG`, 14, 28);
doc.text(`Ngay xuat sao ke: ${new Date().toLocaleDateString('vi-VN')}`, 14, 35);

// Tính tổng thu, tổng chi, số dư để đưa vào PDF
let totalInc = 0;
let totalExp = 0;
transactions.forEach(i => {
if(i.type === 'income') totalInc += i.amount;
else totalExp += i.amount;
});
const finalBal = totalInc - totalExp;

doc.text(`Tong Thu: ${formatVND(totalInc)} | Tong Chi: ${formatVND(totalExp)} | So Du: ${formatVND(finalBal)}`, 14, 42);

// Tạo bảng dữ liệu chi tiết
const tableData = transactions.map((t, index) => [
index + 1,
t.date,
t.description,
t.category,
t.type === 'income' ? 'Thu' : 'Chi',
(t.type === 'income' ? '+' : '-') + t.amount.toLocaleString('vi-VN') + ' VND'
]);

doc.autoTable({
startY: 50,
head: [['STT', 'Ngay', 'Noi dung', 'Danh muc', 'Loai', 'So tien']],
body: tableData,
theme: 'grid',
headStyles: { fillColor: [99, 102, 241] },
styles: { fontSize: 10, cellPadding: 4 }
});

// Tải file PDF về máy
doc.save(`SaoKe_ThuChi_DoanNhutTruong_${new Date().toISOString().slice(0,10)}.pdf`);
alertBox("Đã xuất file PDF sao kê thành công!");
}

// Hàm thông báo thay thế alert()
function alertBox(message) {
document.getElementById('alertMessage').innerText = message;
document.getElementById('customAlert').classList.remove('hidden');
document.getElementById('customAlert').classList.add('flex');
}

function closeAlert() {
document.getElementById('customAlert').classList.remove('flex');
document.getElementById('customAlert').classList.add('hidden');
}

// Chống lỗi XSS cơ bản
function escapeHtml(text) {
const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
return text.replace(/[&<>"']/g, function(m) { return map[m]; });
}
